#include <iostream>
using namespace std;

#define MAX 5

class Stack {
private:
    int arr[MAX];
    int top;

public:
    Stack() {
        top = -1;
    }

    bool isFull() {
        return top == MAX - 1;
    }

    bool isEmpty() {
        return top == -1;
    }

    void push(int val) {
        if (isFull()) {
            cout << "Stack is full!" << endl;
        } else {
            arr[++top] = val;
            cout << val << " pushed into stack." << endl;
        }
    }

    void pop() {
        if (isEmpty()) {
            cout << "Stack is empty!" << endl;
        } else {
            cout << "Popped: " << arr[top--] << endl;
        }
    }

    int peek() {
        if (!isEmpty())
            return arr[top];
        else {
            cout << "Stack is empty!" << endl;
            return -1;
        }
    }

    void display() {
        if (isEmpty()) {
            cout << "Stack is empty!" << endl;
        } else {
            cout << "Stack elements: ";
            for (int i = top; i >= 0; i--) {
                cout << arr[i] << " ";
            }
            cout << endl;
        }
    }
};

int main() {
    Stack s;
    int choice, val;

    do {
        cout << "\n--- Stack Menu ---\n";
        cout << "1. Push\n";
        cout << "2. Pop\n";
        cout << "3. Peek (Top element)\n";
        cout << "4. Check if Empty\n";
        cout << "5. Check if Full\n";
        cout << "6. Display Stack\n";
        cout << "7. Exit\n";
        cout << "Enter your choice: ";
        cin >> choice;

        switch (choice) {
        case 1:
            cout << "Enter value to push: ";
            cin >> val;
            s.push(val);
            break;
        case 2:
            s.pop();
            break;
        case 3:
            val = s.peek();
            if (val != -1)
                cout << "Top element is: " << val << endl;
            break;
        case 4:
            cout << (s.isEmpty() ? "Stack is empty" : "Stack is not empty") << endl;
            break;
        case 5:
            cout << (s.isFull() ? "Stack is full" : "Stack is not full") << endl;
            break;
        case 6:
            s.display();
            break;
        case 7:
            cout << "Exiting program..." << endl;
            break;
        default:
            cout << "Invalid choice!" << endl;
        }
    } while (choice != 7);

    return 0;
}

class STACK {
    char arr[SIZE];
    int top;

public:
    STACK() { top = -1; }

    void push(char val) {
        if (top == SIZE - 1) {
            cout << "Stack is full" << endl;
        } else {
            arr[++top] = val;
        }
    }

    char pop() {
        if (top == -1) {
            return '\0';
        } else {
            return arr[top--];
        }
    }

    bool isEmpty() { return top == -1; }
};

bool isMatching(char open, char close) {
    return (open == '(' && close == ')') ||
           (open == '{' && close == '}') ||
           (open == '[' && close == ']');
}

int main() {
    STACK s;
    string val;

    cout << "Enter expression: ";
    cin >> val;

    for (char ch : val) {
        if (ch == '(' || ch == '{' || ch == '[') {
            s.push(ch);
        } else if (ch == ')' || ch == '}' || ch == ']') {
            if (s.isEmpty()) {
                cout << "Extra closing bracket found: " << ch << endl;
                return 0;
            }
            char popped = s.pop();
            if (!isMatching(popped, ch)) {
                cout << "Unmatched brackets: " << popped << " and " << ch << endl;
                return 0;
            }
        }
    }

    if (s.isEmpty()) {
        cout << "Balanced brackets" << endl;
    } else {
        cout << "Unbalanced brackets: Extra opening bracket(s)" << endl;
    }

    return 0;
}

class Solution {
    vector<long long> s;
    long long minEle;

public:
    Solution() {
        s.clear();
    }

    
    void push(int x) {
        if (s.empty()) {
            s.push_back(x);
            minEle = x;
        }
        else if (x >= minEle) {
            s.push_back(x);
        }
        else {
            
            s.push_back(2LL * x - minEle);
            minEle = x;
        }
    }

    
    void pop() {
        if (s.empty()) return;

        long long t = s.back();
        s.pop_back();

        if (t < minEle) {
            
            minEle = 2LL * minEle - t;
        }
    }

    
    int peek() {
        if (s.empty()) return -1;

        long long t = s.back();
        if (t < minEle) {
            
            return (int)minEle;
        }
        return (int)t;
    }

    
    int getMin() {
        if (s.empty()) return -1;
        return (int)minEle;
    }
};

#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    bool isValid(string expr) {
        stack<char> s;

        for (char c : expr) {
            if (c == '(' || c == '{' || c == '[') {
                s.push(c);
            }
            else if (c == ')' || c == '}' || c == ']') {
                if (s.empty()) return false;
                char top = s.top();
                s.pop();
                if (!isMatching(top, c)) return false;
            }
        }
        return s.empty();
    }

private:
    bool isMatching(char open, char close) {
        return (open == '(' && close == ')') ||
               (open == '{' && close == '}') ||
               (open == '[' && close == ']');
    }
};

int main() {
    Solution sol;
    string expr;
    cout << "Enter expression: ";
    cin >> expr;

    if (sol.isValid(expr))
        cout << "Balanced Expression" << endl;
    else
        cout << "Not Balanced" << endl;

    return 0;
}

class Solution {
  public:
    vector<int> nextLargerElement(vector<int>& arr) 
    {
        int n = arr.size();
        vector<int> result(n, -1);  
        stack<int> s;              
        
        for (int i = 0; i < n; i++) {
            while (!s.empty() && arr[i] > arr[s.top()]) {
                result[s.top()] = arr[i];
                s.pop();
            }
            s.push(i);
        }
        return result;
    }
};

#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        int n = temperatures.size();
        vector<int> answer(n, 0);
        stack<int> s;

        for (int i = 0; i < n; i++) {
            while (!s.empty() && temperatures[i] > temperatures[s.top()]) {
                int index = s.top();
                s.pop();
                answer[index] = i - index;
            }
            s.push(i);
        }

        return answer;
    }
};

    vector<int> res = sol.dailyTemperatures(temps);

    for (int x : res) cout << x << " ";
    cout << endl;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    bool canBeSorted(vector<int>& AA) {
        int NN = AA.size();
        vector<int> BB;       
        stack<int> SS;      

       
        vector<int> sortedAA = AA;
        sort(sortedAA.begin(), sortedAA.end());

        int targetIndex = 0;

        
        for (int i = 0; i < NN; i++) {
            SS.push(AA[i]);  // operation 1

            
            while (!SS.empty() && SS.top() == sortedAA[targetIndex]) {
                BB.push_back(SS.top()); 
                SS.pop();
                targetIndex++;
            }
        }

        
        while (!SS.empty() && SS.top() == sortedAA[targetIndex]) {
            BB.push_back(SS.top());
            SS.pop();
            targetIndex++;
        }

       
        return (targetIndex == NN);
    }
};

int main() {
    Solution sol;

    vector<int> AA1 = {3, 1, 2};
    cout << (sol.canBeSorted(AA1) ? "YES" : "NO") << endl;

    vector<int> AA2 = {3, 2, 1};
    cout << (sol.canBeSorted(AA2) ? "YES" : "NO") << endl;

    return 0;
}
