#include <iostream>
using namespace std;

struct DNode {
    int data;
    DNode* next;
    DNode* prev;
    DNode(int val) : data(val), next(nullptr), prev(nullptr) {}
};

class DoublyLinkedList {
public:
    DNode* head;

    DoublyLinkedList() : head(nullptr) {}

    void insertAtBeginning(int val) {
        DNode* newNode = new DNode(val);
        if (!head) {
            head = newNode;
            return;
        }
        newNode->next = head;
        head->prev = newNode;
        head = newNode;
    }

    void insertAtEnd(int val) {
        DNode* newNode = new DNode(val);
        if (!head) {
            head = newNode;
            return;
        }
        DNode* temp = head;
        while (temp->next)
            temp = temp->next;

        temp->next = newNode;
        newNode->prev = temp;
    }

    void insertAfter(int key, int val) {
        if (!head) return;
        DNode* temp = head;
        while (temp && temp->data != key)
            temp = temp->next;

        if (!temp) {
            cout << "Node not found.\n";
            return;
        }

        DNode* newNode = new DNode(val);
        newNode->next = temp->next;
        newNode->prev = temp;

        if (temp->next)
            temp->next->prev = newNode;

        temp->next = newNode;
    }

    void insertBefore(int key, int val) {
        if (!head) return;

        if (head->data == key) {
            insertAtBeginning(val);
            return;
        }

        DNode* temp = head->next;
        while (temp && temp->data != key)
            temp = temp->next;

        if (!temp) {
            cout << "Node not found.\n";
            return;
        }

        DNode* newNode = new DNode(val);
        newNode->next = temp;
        newNode->prev = temp->prev;
        temp->prev->next = newNode;
        temp->prev = newNode;
    }

    void display() {
        if (!head) {
            cout << "(empty)\n";
            return;
        }
        DNode* temp = head;
        while (temp) {
            cout << temp->data << " ";
            temp = temp->next;
        }
        cout << "\n";
    }
};

    DNode* prev;

class DoublyLinkedList {
public:
    DNode* head;

    DoublyLinkedList() : head(nullptr) {}

    void insertAtEnd(int val) {
        DNode* n = new DNode(val);
        if (!head) {
            head = n;
            return;
        }
        DNode* temp = head;
        while (temp->next) temp = temp->next;
        temp->next = n;
        n->prev = temp;
    }

    void deleteNode(int key) {
        if (!head) {
            cout << "List empty.\n";
            return;
        }

        DNode* temp = head;

        if (head->data == key) {
            head = head->next;
            if (head) head->prev = nullptr;
            delete temp;
            cout << "Node " << key << " deleted.\n";
            return;
        }

        while (temp && temp->data != key)
            temp = temp->next;

        if (!temp) {
            cout << "Node " << key << " not found.\n";
            return;
        }

        if (!temp->next) {
            temp->prev->next = nullptr;
            delete temp;
            cout << "Node " << key << " deleted.\n";
            return;
        }

        temp->prev->next = temp->next;
        temp->next->prev = temp->prev;
        delete temp;

        cout << "Node " << key << " deleted.\n";
    }

    void display() {
        if (!head) {
            cout << "(empty)\n";
            return;
        }
        DNode* t = head;
        while (t) {
            cout << t->data << " ";

QUESTION 4. Write a program to check if a Doubly Linked List of characters is palindrome or not.

Example:
Input: L → E → V → E → L
Output: True
====================================================================================================================================
SOLUTION:
====================================================================================================================================
#include <iostream>
using namespace std;

struct Node {
    char data;
    Node* next;
    Node* prev;
    Node(char val) : data(val), next(nullptr), prev(nullptr) {}
};

class DoublyLinkedList {
public:
    Node* head;

    DoublyLinkedList() : head(nullptr) {}

    
    void insert(char val) {
        Node* newNode = new Node(val);
        if (!head) {
            head = newNode;
            return;
        }
        Node* temp = head;
        while (temp->next)
            temp = temp->next;
        temp->next = newNode;
        newNode->prev = temp;
    }

    bool isPalindrome() {
        if (!head || !head->next) return true;

        Node* tail = head;
        while (tail->next)
            tail = tail->next;

        Node* start = head;
        Node* end = tail;

        while (start != end && start->prev != end) {
            if (start->data != end->data)
                return false;

            start = start->next;
            end = end->prev;
        }
        return true;
    }

    void display() {
        Node* temp = head;
        while (temp) {
            cout << temp->data << " ";
            temp = temp->next;
        }
        cout << endl;
    }
};

int main() {

    cout << "Example 1:" << endl;
    DoublyLinkedList dll;

    dll.insert('L');
    dll.insert('E');
    dll.insert('V');
    dll.insert('E');
    dll.insert('L');

    cout << "Doubly Linked List: ";
    dll.display();

    if (dll.isPalindrome())
        cout << "Output: True (Palindrome)" << endl;
    else
        cout << "Output: False (Not Palindrome)" << endl;



    cout << "\nExample 2:" << endl;
    DoublyLinkedList dll2;

    dll2.insert('A');
    dll2.insert('B');
    dll2.insert('C');
    dll2.insert('B');

    cout << "Doubly Linked List: ";
    dll2.display();

    if (dll2.isPalindrome())
        cout << "Output: True (Palindrome)" << endl;
    else
        cout << "Output: False (Not Palindrome)" << endl;

    return 0;
}

QUESTION 5. Write a program to check if a linked list is Circular Linked List or not.

Example:
Linked List: 2 → 4 → 6 → 7 → 5
(Output True if last node points back to head)
Output: True
====================================================================================================================================
SOLUTION:
====================================================================================================================================
#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* next;
    Node(int val) : data(val), next(nullptr) {}
};

class LinkedList {
    Node* head;

public:
    LinkedList() : head(nullptr) {}

    void insert(int val) {
        Node* newNode = new Node(val);

        if (!head) {
            head = newNode;
            return;
        }

        Node* temp = head;
        while (temp->next != nullptr)
            temp = temp->next;

        temp->next = newNode;
    }

    void makeCircular() {
        if (!head) return;

        Node* temp = head;
        while (temp->next != nullptr)
            temp = temp->next;

        temp->next = head;
    }

    
    bool isCircular() {
        if (!head) return true;

        Node* temp = head->next;

        while (temp != nullptr && temp != head) {
            temp = temp->next;
        }

        return (temp == head);
    }
};

int main() {

    cout << "Example 1:" << endl;
    LinkedList list1;

    list1.insert(2);
    list1.insert(4);
    list1.insert(6);
    list1.insert(7);
    list1.insert(5);

    
    list1.makeCircular();

    cout << "Linked List: 2 → 4 → 6 → 7 → 5" << endl;
    cout << "Output: " << (list1.isCircular() ? "True" : "False") << endl;



    cout << "\nExample 2:" << endl;
    LinkedList list2;

    list2.insert(10);
    list2.insert(20);
    list2.insert(30);

    
    cout << "Linked List: 10 → 20 → 30" << endl;
